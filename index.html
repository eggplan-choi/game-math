[활동지 01] 2D 물리 엔진의 기초: 화면과 중력, 그리고 점프

🎯 학습 목표

  1. 컴퓨터 화면의 좌표계와 게임 루프(Game Loop)의 동작 원리를 이해한다.
  2. 중력 가속도 수식을 코드로 모델링하여 캐릭터가 포물선 운동을 하도록 만든다.
  3. 바닥 착지 조건을 제어하여 무한 점프 버그를 방지한다.

1. 게임 루프와 화면 좌표계

컴퓨터 모니터는 수학 시간에 배우는 데카르트 좌표계와 달리 화면의 왼쪽 위가 원점 $(0, 0)$ 입니다.

  • $x$ 축: 오른쪽으로 갈수록 값이 커집니다 ($+$).
  • $y$ 축: 아래로 내려갈수록 값이 커집니다 ($+$).

📌 Step 1: 기본 창 띄우고 좌우로 움직이기

🐍 step1_basic_loop.py
# [1차시 Step 1] 컴퓨터 화면 좌표계와 게임 루프의 기본 구조
import pygame
import sys

# 1. 화면 및 시계(FPS) 초기화
pygame.init()
SCREEN_WIDTH = 800     # 화면의 가로 픽셀 길이 (x축의 최대 범위)
SCREEN_HEIGHT = 450    # 화면의 세로 픽셀 길이 (y축의 최대 범위)
screen = pygame.display.set_mode((SCREEN_WIDTH, SCREEN_HEIGHT))
pygame.display.set_caption("1차시: 플레이어 이동과 기본 루프")
clock = pygame.time.Clock()

# 2. 플레이어의 초기 상태 정의 (수학의 점 (x, y)에 크기를 부여)
player_x = 100         # 시작 x좌표 (화면 왼쪽에서 100px)
player_y = 200         # 시작 y좌표 (화면 위쪽에서 200px)
player_size = 40       # 한 변의 길이가 40px인 정사각형 캐릭터
speed = 5              # 1프레임당 이동할 변위 Delta x (픽셀 단위의 수평 속도)

# --- 메인 게임 루프 (Game Loop: 초당 60회 반복 수행되는 시간의 흐름) ---
running = True
while running:
    # (1) 이벤트 수집: 사용자의 입력(마우스, 키보드, 창 닫기 등)을 감지
    for event in pygame.event.get():
        if event.type == pygame.QUIT:
            running = False

    # (2) 입력에 따른 좌표 갱신 (위치 함수의 이산적 변화: x(t+1) = x(t) + v)
    keys = pygame.key.get_pressed()
    if keys[pygame.K_LEFT]:
        player_x -= speed  # 왼쪽 이동: x좌표 감소 (음의 방향 이동)
    if keys[pygame.K_RIGHT]:
        player_x += speed  # 오른쪽 이동: x좌표 증가 (양의 방향 이동)

    # (3) 화면 밖 이탈 방지 (수학적 닫힌 구간 클램핑: 0 <= player_x <= MAX_X)
    # min과 max 함수를 합성하여 좌표를 화면 영역 내부 [0, SCREEN_WIDTH - 크기]로 제한
    player_x = max(0, min(player_x, SCREEN_WIDTH - player_size))

    # (4) 렌더링 (화면 출력: 매 순간마다 이전 그림을 지우고 새 위치에 그리기)
    screen.fill((30, 30, 30))  # 어두운 단색으로 화면을 덮어 이전 잔상을 지움
    # 모니터 좌표계 상에 사각형을 그림: Rect(x, y, width, height)
    pygame.draw.rect(screen, (0, 200, 255), (player_x, player_y, player_size, player_size))
    pygame.display.flip()

    # 1초에 정확히 60번 루프를 돌도록 제어 (dt = 1/60초 고정)
    clock.tick(60)

pygame.quit()
sys.exit()

2. 중력 가속도와 점프 (물리 법칙 도입)

$$v_y \leftarrow v_y + g$$
$$y \leftarrow y + v_y$$

# [1차시 Step 2] 등가속도 운동 점화식을 이용한 자유낙하와 포물선 점프 모델링
import pygame
import sys

pygame.init()
SCREEN_WIDTH = 800
SCREEN_HEIGHT = 450
screen = pygame.display.set_mode((SCREEN_WIDTH, SCREEN_HEIGHT))
pygame.display.set_caption("1차시: 중력과 점프 물리 엔진")
clock = pygame.time.Clock()

player_x = 100
player_y = 100
player_size = 40
speed = 5

# --- 물리 엔진 파라미터 (미적분학의 이산적 오일러 적분 모델) ---
velocity_y = 0.0      # 세로 방향 현재 속도 v_y (초기 속도는 0)
gravity = 0.8         # 중력 가속도 g (매 프레임마다 속도에 더해지는 상수: 등가속도)
jump_power = -14.0    # 점프 초기 속도 v_0 (컴퓨터 화면은 위쪽이 음수이므로 음의 부호 부여)
is_grounded = False   # 바닥 착지 여부를 나타내는 불리언(Boolean) 상태 플래그

# 지면의 y좌표 방정식: y = FLOOR_Y
FLOOR_Y = 380

running = True
while running:
    for event in pygame.event.get():
        if event.type == pygame.QUIT:
            running = False

        # [점프 트리거 조건] 바닥에 닿아 있을 때만(is_grounded == True) 위쪽 초기속도를 부여
        # 이 조건이 없으면 공중에서 계속 점프할 수 있는 무한 다단 점프 버그가 발생함
        if event.type == pygame.KEYDOWN:
            if event.key == pygame.K_SPACE and is_grounded:
                velocity_y = jump_power   # 상향 초기속도 부여 (포물선 운동의 시작점)
                is_grounded = False       # 점프 즉시 공중에 뜬 상태로 전환

    # 1. 수평 등속도 이동 처리
    keys = pygame.key.get_pressed()
    if keys[pygame.K_LEFT]: player_x -= speed
    if keys[pygame.K_RIGHT]: player_x += speed
    player_x = max(0, min(player_x, SCREEN_WIDTH - player_size))

    # 2. 물리 연산 (수열의 점화식을 매 프레임 누적 계산)
    # 속도의 변화: v(t+1) = v(t) + g (가속도를 적분하여 속도 갱신)
    velocity_y += gravity
    # 위치의 변화: y(t+1) = y(t) + v(t+1) (속도를 적분하여 위치 갱신)
    player_y += velocity_y

    # 3. 바닥 충돌 및 수직항력(Normal Force) 시뮬레이션
    # 플레이어의 발밑 y좌표(player_y + player_size)가 바닥(FLOOR_Y)보다 아래로 내려갔을 때

class MovingPlatform:
    """수평 닫힌 구간 [start_x, end_x] 사이를 일정한 속도로 왕복 운동하는 발판"""
    def __init__(self, x, y, width, height, move_range_x, move_speed):
        self.rect = pygame.Rect(x, y, width, height)
        self.start_x = x
        self.end_x = x + move_range_x
        self.speed_x = move_speed

    def update(self):
        # 발판의 위치 갱신: x(t+1) = x(t) + v_platform
        self.rect.x += self.speed_x
        # 경계 도달 시 속도 부호 반전 (v <- -v)
        if self.rect.x >= self.end_x or self.rect.x <= self.start_x:
            self.speed_x *= -1
            return self.speed_x * -1  # 방향 반전 직전의 1프레임 변위 반환
        return self.speed_x

def generate_beep_sound(freq_start, freq_end, duration_ms, wave_type="square"):
    """
    [삼각함수 음파 생성기]
    주파수 f를 가진 소리는 사인파 y = A * sin(2 * pi * f * t)로 표현됨
    외부 wav 파일 없이 수학적 파동 함수를 직접 연산하여 8비트 사운드 버퍼 생성
    """
    sample_rate = 44100  # 1초당 샘플 수 (44.1kHz)
    n_samples = int(sample_rate * (duration_ms / 1000.0))
    buf = array.array('h')
    for i in range(n_samples):
        t = i / n_samples
        freq = freq_start + (freq_end - freq_start) * t  # 시간에 따른 주파수 선형 보간
        phase = 2.0 * math.pi * freq * (i / sample_rate) # 위상각 세타 계산
        val = 16000 if math.sin(phase) >= 0 else -16000 # 구형파(Square wave) 양수/음수 이진화
        decay = (1.0 - t)                                # 선형 감쇠 (음량이 점차 줄어듦)
        buf.append(int(val * decay))
    return pygame.mixer.Sound(buf)

class Particle:
    """[극좌표계 -> 직교좌표계 변환] 임의의 각도(세타)와 속력(v)으로 튀어나가는 미세 파티클"""
    def __init__(self, x, y, color):
        self.x, self.y = x, y
        self.color = color
        self.size = random.randint(6, 12)
        angle = random.uniform(0, 2 * math.pi)  # 0부터 2pi 사이의 무작위 라디안 각도 세타
        speed = random.uniform(3, 8)            # 스칼라 속력 v

        self.vx = math.cos(angle) * speed
        self.vy = math.sin(angle) * speed - 2   # 살짝 위쪽으로 솟구치도록 y속도 오프셋
        self.gravity = 0.4
        self.life = 1.0  # 생명력 1.0에서 시작하여 0.0으로 감소

    def update(self):
        self.vy += self.gravity  # 파티클에도 작은 중력 가속도 적용
        self.x += self.vx
        self.y += self.vy
        self.life -= 0.03        # 매 프레임 수명 감소
        self.size = max(0, self.size - 0.15)  # 시간에 따라 크기 축소 (닮음비 축소)

    def draw(self, surface, ox, oy):
        if self.life > 0 and self.size > 0:
            pygame.draw.rect(surface, self.color, (int(self.x + ox), int(self.y + oy), int(self.size), int(self.size)))

def find_a_star_path(start_pos, target_pos):
    """
    [A* 알고리즘의 수학적 평가 함수]
    f(n) = g(n) + h(n)
    - g(n): 시작 노드에서 현재 노드 n까지 실제 이동한 거리
    - h(n): 현재 노드 n에서 목적지까지의 추정 거리 (L1 노름 / 맨해튼 거리: |dr| + |dc|)
    """
    start_c, start_r = int(start_pos[0] // TILE_SIZE), int(start_pos[1] // TILE_SIZE)
    target_c, target_r = int(target_pos[0] // TILE_SIZE), int(target_pos[1] // TILE_SIZE)

    # 목적지가 맵 밖이거나 벽 내부이면 경로 탐색 불가능
    if not (0 <= target_r < grid_rows and 0 <= target_c < grid_cols): return []
    if collision_grid[target_r][target_c]: return []

    open_set = []
    # 힙 튜플: (총비용 f, 실제비용 g, (행 r, 열 c))
    heapq.heappush(open_set, (0, 0, (start_r, start_c)))
    came_from = {}
    g_score = {(start_r, start_c): 0}

    while open_set:
        _, current_g, current = heapq.heappop(open_set)

        # 목적지에 도달한 경우 경로 역추적 (Backtracking)
        if current == (target_r, target_c):
            path = []
            curr = current
            while curr in came_from:
                # 격자 좌표를 픽셀 중심 좌표로 변환하여 저장
                path.append((curr[1] * TILE_SIZE, curr[0] * TILE_SIZE))
                curr = came_from[curr]
            path.reverse()  # 시작점 -> 끝점 순서로 뒤집음
            return path

        cr, cc = current
        # 상, 하, 좌, 우 4방향 탐색 단위 벡터
        for dr, dc in [(-1, 0), (1, 0), (0, -1), (0, 1)]:
            nr, nc = cr + dr, cc + dc
            if 0 <= nr < grid_rows and 0 <= nc < grid_cols:
                if collision_grid[nr][nc]: continue  # 장애물 격자는 건너뜀
                tent_g = current_g + 1              # 등간격 격자이므로 1칸 이동 비용은 1
                if (nr, nc) not in g_score or tent_g < g_score[(nr, nc)]:
                    g_score[(nr, nc)] = tent_g
                    # 맨해튼 거리 휴리스틱: h = |r1 - r2| + |c1 - c2|
                    h_score = abs(nr - target_r) + abs(nc - target_c)
                    f_score = tent_g + h_score
                    heapq.heappush(open_set, (f_score, tent_g, (nr, nc)))
                    came_from[(nr, nc)] = current
    return []
