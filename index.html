1. 게임 루프와 화면 좌표계

컴퓨터 모니터는 수학 시간에 배우는 데카르트 좌표계와 달리 화면의 왼쪽 위가 원점 $ (0, 0) $ 입니다.

  • $ x $ 축: 오른쪽으로 갈수록 값이 커집니다 ($+$).
  • $ y $ 축: 아래로 내려갈수록 값이 커집니다 ($+$).

📌 Step 1: 기본 창 띄우고 좌우로 움직이기

🐍 step1_basic_loop.py
# [1차시 Step 1] 컴퓨터 화면 좌표계와 게임 루프의 기본 구조
import pygame
import sys

# 1. 파이게임 라이브러리와 화면 크기(해상도) 초기화
pygame.init()          # 파이게임의 다양한 기능(그래픽, 소리 등)을 사용하기 위해 준비합니다.
SCREEN_WIDTH = 800     # 화면의 가로 픽셀 길이 (x축의 최대 범위)
SCREEN_HEIGHT = 450    # 화면의 세로 픽셀 길이 (y축의 최대 범위)
screen = pygame.display.set_mode((SCREEN_WIDTH, SCREEN_HEIGHT)) # 설정한 크기의 창을 생성합니다.
pygame.display.set_caption("1차시: 플레이어 이동과 기본 루프") # 창의 제목 표시줄을 설정합니다.
clock = pygame.time.Clock() # 초당 프레임 수(FPS)를 조절하기 위한 시계 객체를 만듭니다.

# 2. 플레이어의 초기 상태 정의 (수학의 점 (x, y)에 크기를 부여)
player_x = 100         # 캐릭터의 시작 x좌표 (화면 왼쪽에서 100px)
player_y = 200         # 캐릭터의 시작 y좌표 (화면 위쪽에서 200px)
player_size = 40       # 캐릭터의 가로/세로 길이 (40px 크기의 정사각형)
speed = 5              # 1프레임당 이동할 거리 (픽셀 단위의 수평 속도)

# --- 메인 게임 루프 (Game Loop: 초당 60회 반복 수행되는 무한 루프) ---
running = True
while running:
    # (1) 이벤트 수집: 창 닫기 버튼(X)을 누르면 게임 루프를 종료합니다.
    for event in pygame.event.get():
        if event.type == pygame.QUIT:
            running = False

    # (2) 입력에 따른 좌표 갱신: 키보드의 왼쪽/오른쪽 화살표 키가 눌렸는지 확인합니다.
    keys = pygame.key.get_pressed()
    if keys[pygame.K_LEFT]:
        player_x -= speed  # 왼쪽 이동: x좌표 감소 (원점 방향으로 이동)
    if keys[pygame.K_RIGHT]:
        player_x += speed  # 오른쪽 이동: x좌표 증가 (원점 반대 방향으로 이동)

    # (3) 화면 밖 이탈 방지: 캐릭터가 화면 양옆으로 사라지지 않도록 x좌표의 범위를 제한합니다.
    # 수학적으로 표현하면 0 <= player_x <= SCREEN_WIDTH - player_size 와 같습니다.
    player_x = max(0, min(player_x, SCREEN_WIDTH - player_size))

    # (4) 렌더링 (화면 출력): 매 프레임마다 이전 그림을 지우고 새로운 위치에 캐릭터를 다시 그립니다.
    screen.fill((30, 30, 30))  # 어두운 회색으로 화면 전체를 덮어 이전 잔상을 지웁니다. RGB(30, 30, 30)
    # 지정한 위치(player_x, player_y)에 하늘색 RGB(0, 200, 255) 사각형을 그립니다.
    pygame.draw.rect(screen, (0, 200, 255), (player_x, player_y, player_size, player_size))
    pygame.display.flip()      # 메모리에 그린 화면을 모니터에 실제로 출력(업데이트)합니다.

    # 1초에 정확히 60번 루프를 돌도록 제어합니다 (dt = 1/60초 고정).
    clock.tick(60)

# 무한 루프가 종료되면 파이게임을 종료하고 프로그램을 닫습니다.
pygame.quit()
sys.exit()

2. 중력 가속도와 점프 (물리 법칙 도입)

$$v_y \leftarrow v_y + g$$
$$y \leftarrow y + v_y$$

# [1차시 Step 2] 등가속도 운동 점화식을 이용한 자유낙하와 포물선 점프 모델링
import pygame
import sys

pygame.init()
SCREEN_WIDTH = 800
SCREEN_HEIGHT = 450
screen = pygame.display.set_mode((SCREEN_WIDTH, SCREEN_HEIGHT))
pygame.display.set_caption("1차시: 중력과 점프 물리 엔진")
clock = pygame.time.Clock()

player_x = 100
player_y = 100
player_size = 40
speed = 5

# --- 물리 엔진 파라미터 (자연스러운 포물선을 만들기 위한 변수들) ---
velocity_y = 0.0      # 세로 방향 현재 속도 (현재 아래쪽으로 얼마나 빠르게 떨어지는가)
gravity = 0.8         # 중력 가속도 (매 프레임마다 속도에 더해지는 일정한 값, 속도를 증가시킴)
jump_power = -14.0    # 점프 초기 속도 (점프 시 위로 솟구치는 힘. 화면 위쪽 방향은 y가 감소하므로 음수입니다)
is_grounded = False   # 현재 바닥에 닿아있는지(착지 상태인지) 여부를 저장하는 불리언(True/False) 플래그

FLOOR_Y = 380         # 바닥의 y좌표 위치 (이 선 아래로는 떨어지지 않습니다)

running = True
while running:
    for event in pygame.event.get():
        if event.type == pygame.QUIT:
            running = False

        # 스페이스바를 누르고 있고(KEYDOWN), 동시에 바닥에 닿아있을 때(is_grounded)만 점프를 허용합니다.
        # 바닥에 있다는 조건이 없으면 공중에서 무한 점프를 하게 됩니다.
        if event.type == pygame.KEYDOWN:
            if event.key == pygame.K_SPACE and is_grounded:
                velocity_y = jump_power   # 강한 위쪽(-) 속도를 부여하여 위로 솟구치게 합니다.

# 스스로 왕복 이동하는 발판 객체를 만들기 위한 클래스(Class) 청사진입니다.
class MovingPlatform:
    def __init__(self, x, y, width, height, move_range_x, move_speed):
        self.rect = pygame.Rect(x, y, width, height)
        self.start_x = x                       # 시작 x 좌표 (왕복의 왼쪽 끝점)
        self.end_x = x + move_range_x          # 도착 x 좌표 (왕복의 오른쪽 끝점)
        self.speed_x = move_speed              # 이동 속도

    def update(self):
        # 매 프레임마다 speed_x만큼 스스로 이동합니다.
        self.rect.x += self.speed_x
        # 지정된 범위를 벗어나면 속도의 부호를 반대로 뒤집어(x -1) 이동 방향을 반전시킵니다.
        if self.rect.x >= self.end_x or self.rect.x <= self.start_x:
            self.speed_x *= -1
            return self.speed_x * -1 # 방향이 바뀌는 순간의 이동 거리를 반환
        return self.speed_x # 플레이어를 같이 싣고 가기 위해 이번 프레임의 이동 거리를 반환

import pygame
import sys

import array

# 오디오 버퍼 설정을 초기화하여 사운드 지연 현상을 방지합니다.

# 외부 mp3 파일 없이 수학적인 파동(사인파, 사각파)을 배열에 채워 넣어 8비트 효과음을 합성하는 함수입니다.
def generate_beep_sound(freq_start, freq_end, duration_ms, wave_type="square"):
    sample_rate = 44100 # 초당 추출되는 소리 샘플 개수
    n_samples = int(sample_rate * (duration_ms / 1000.0))
    buf = array.array('h')
    for i in range(n_samples):
        t = i / n_samples
        # 시작 주파수에서 끝 주파수로 음의 높낮이가 서서히 변하도록(Glissando) 보간합니다.
        freq = freq_start + (freq_end - freq_start) * t
        phase = 2.0 * math.pi * freq * (i / sample_rate)
        # 수학 함수를 이용해 파형 생성 (음수 양수를 오가며 진동)
        val = 16000 if math.sin(phase) >= 0 else -16000
        decay = (1.0 - t) # 시간이 지날수록 소리 크기(진폭)가 점점 작아짐
        buf.append(int(val * decay))
    return pygame.mixer.Sound(buf)

# 200Hz에서 시작해 600Hz로 0.12초간 소리가 올라가는 점프 효과음을 생성합니다.

# 화면에 흩뿌려지는 장식용 조각(파티클) 클래스
class Particle:
    def __init__(self, x, y, color):
        self.x, self.y = x, y
        self.color = color
        self.size = random.randint(6, 12) # 파티클의 크기를 무작위로 설정
        
        # [수학 적용] 삼각함수와 각도(라디안)를 이용하여 전 방위 360도로 흩어지는 x, y 속도 벡터를 계산합니다.
        angle = random.uniform(0, 2 * math.pi)
        speed = random.uniform(3, 8)
        self.vx = math.cos(angle) * speed
        self.vy = math.sin(angle) * speed - 2
        
        self.gravity = 0.4
        self.life = 1.0 # 투명도를 관리하는 수명 변수 (1.0 = 100%)

    def update(self):
        # 파티클도 중력을 받아 포물선을 그리며 떨어집니다.
        self.vy += self.gravity
        self.x += self.vx
        self.y += self.vy
        self.life -= 0.03                 # 매 프레임마다 수명이 깎임
        self.size = max(0, self.size - 0.15) # 파티클 크기가 점점 줄어들게 연출

    def draw(self, surface, ox, oy):
        # 수명과 크기가 0보다 클 때만 화면에 그립니다. 
        if self.life > 0 and self.size > 0:
            # (ox, oy)는 화면 흔들림(카메라 지진) 값을 파티클에도 반영하기 위한 오프셋입니다.
            pygame.draw.rect(surface, self.color, (int(self.x + ox), int(self.y + oy), int(self.size), int(self.size)))

[활동지 08] 특수 아이템 II: 비행 워프와 $ A^* $ (A-Star) 경로 탐색

1. $ A^* $ (에이스타) 알고리즘의 구동 원리

복잡한 맵에서 목적지까지 이동할 때 단순히 목표물이 있는 직선 방향으로만 향하면 벽에 막혀 오도가도 못하는 상황이 발생합니다. 이를 지능적으로 해결하기 위해 A* 알고리즘을 사용합니다. 이 기법은 현재 위치에서 목표까지의 '예상 거리'와 출발지부터 현재까지 '이동한 누적 거리'를 종합하여 장애물을 똑똑하게 피해 가는 최적의 우회 경로를 찾아냅니다.

A*알고리즘의 구동 원리 유튜브 설명 영상

📌 가로 7, 세로 7 격자에서의 경로 탐색 예시

총 49개의 칸으로 구성된 7 by 7 맵이 있고, 캐릭터는 왼쪽 위 구석인 출발점 $ (0, 0) $ 에서 오른쪽 아래 구석인 도착점 $ (6, 6) $ 까지 살아서 가야 한다고 가정해 보겠습니다. 그런데 화면 중앙에 거대한 벽(장애물)이 길을 막고 있는 상황입니다.

🚀 A* 알고리즘 실시간 경로 탐색 시뮬레이터 빈 공간을 클릭하여 도착점(G)을 변경해 보세요!
■ 탐색 완료 ■ 탐색 대기 ■ 최단 경로 ■ 장애물(벽)

💡 탐색 알고리즘의 진행 과정

  1. 시작점 평가: 출발점 $ (0, 0) $ 에서 상하좌우로 붙어있는 타일들에 가상으로 이동했다고 가정해 봅니다.
  2. 점수 매기기 (F = G + H): 평가할 각 타일마다 점수를 계산합니다. 출발점으로부터 실제로 이동한 칸 수(G)와, 현재 칸에서 도착점 $ (6, 6) $ 까지 남은 예상 직선 거리(H, 휴리스틱)를 더해 총점(F)을 구합니다.
  3. 최적의 길 선택: 계산된 총점(F)이 가장 낮은, 즉 도착점에 가장 빨리 도달할 것으로 예상되는 유망한 타일을 다음 이동 경로로 확정 짓습니다.
  4. 장애물 우회: 점수가 낮아 순조롭게 진행하던 중 중앙의 벽을 만나면, 벽을 통과할 수 없으므로 탐색을 멈춥니다. 그 대신 아까 점수를 매겨두었던 다른 우회로(장애물 가장자리 타일들) 중 가장 점수가 낮은 차선책을 골라 탐색을 이어 나갑니다.
  5. 경로 완성: 도착점 $ (6, 6) $ 을 찾아내면, 여태까지 기록해 두었던 발자취(타일들)를 역으로 거슬러 올라가 최종적인 비행 궤적 배열을 완성합니다.

import sys
import os
import csv
import heapq  # A* 탐색에서 '가장 점수가 낮은 경로'를 효율적으로 뽑아내는 우선순위 큐 수학 모듈입니다.

pygame.init()
SCREEN_WIDTH = 800
SCREEN_HEIGHT = 450
screen = pygame.display.set_mode((SCREEN_WIDTH, SCREEN_HEIGHT))
pygame.display.set_caption("8차시: A* 경로 탐색 기반 비행 워프")
clock = pygame.time.Clock()
font_ui = pygame.font.SysFont(None, 26)
font_prompt = pygame.font.SysFont(None, 34)

TILE_SIZE = 40
platforms, hazards, warp_items = [], [], []
goal_rect = None
spawn_point = (100, 100)

grid_rows = grid_cols = 0
collision_grid = [] # A* 알고리즘이 맵의 어느 곳이 막혀있는 벽인지 알 수 있도록 변환해두는 2차원 리스트

def load_map_from_csv(filepath):
    global platforms, hazards, warp_items, goal_rect, spawn_point, grid_rows, grid_cols, collision_grid
    platforms.clear()
    hazards.clear()
    warp_items.clear()
    goal_rect = None
    spawn_point = (100, 100)

    if not os.path.exists(filepath):
        platforms.append(pygame.Rect(0, 380, SCREEN_WIDTH * 2, 70))
        return

    with open(filepath, mode="r", encoding="utf-8-sig") as f:
        reader = list(csv.reader(f))
        grid_rows = len(reader)
        grid_cols = len(reader[0]) if grid_rows > 0 else 0
        
        # 맵 크기만큼 비어있는(False) 충돌 판정용 배열 격자를 생성합니다.
        collision_grid = [[False for _ in range(grid_cols)] for _ in range(grid_rows)]

        for r, row in enumerate(reader):
            for c, raw_val in enumerate(row):
                val = raw_val.strip()
                x, y = c * TILE_SIZE, r * TILE_SIZE
                
                # 타일이 벽(1)이거나 가시 함정(2)이면, A* 알고리즘이 통과할 수 없는 장애물(True)로 기록해 둡니다.
                if val == "1":
                    platforms.append(pygame.Rect(x, y, TILE_SIZE, TILE_SIZE))
                    collision_grid[r][c] = True
                elif val == "2":
                    hazards.append(pygame.Rect(x, y, TILE_SIZE, TILE_SIZE))
                    collision_grid[r][c] = True  
                elif val.upper() == "W":
                    warp_items.append(pygame.Rect(x + 10, y + 10, 20, 20)) # 워프 주황색 아이템 객체
                elif val.upper() == "P": spawn_point = (x, y)
                elif val.upper() == "G": goal_rect = pygame.Rect(x, y, TILE_SIZE, TILE_SIZE * 2)

player = pygame.Rect(0, 0, TILE_SIZE, TILE_SIZE)
speed = 6
velocity_x = velocity_y = 0.0
gravity = 0.8
jump_power = -15.0
is_grounded = False

# [워프 비행 관련 상태 플래그 변수]
has_warp_ability = False  # 워프 아이템(W)을 획득하여 '마우스 클릭을 기다리는 조준 대기 상태'인가?
is_flying = False         # 마우스를 클릭해서 목적지 좌표까지 '경로를 따라 날아가는 중'인가?
fly_path = []             # A* 알고리즘이 계산해낸 꺾이는 좌표점들의 리스트 배열
fly_speed = 10.0          # 비행하여 돌진할 때의 속도

def get_padded_rect(rect, pad):
    """사각 충돌 판정 여유폭 적용 함수"""
    return pygame.Rect(rect.x + pad, rect.y + pad, rect.width - 2 * pad, rect.height - 2 * pad)

def reset_stage():
    """
    [워프 스테이지 및 아이템 완전 초기화 함수]
    """
    global velocity_x, velocity_y, has_warp_ability, is_flying, fly_path
    load_map_from_csv("stage_warp.csv")
    player.x, player.y = spawn_point
    velocity_x = 0.0
    velocity_y = 0.0
    has_warp_ability = False # 사망하면 워프 능력도 몰수
    is_flying = False
    fly_path.clear()

# [핵심] A* 최단 우회 경로 탐색 알고리즘
def find_a_star_path(start_pos, target_pos):
    # 픽셀 단위 좌표를 타일(칸) 단위 인덱스로 바꿉니다 (40px로 나누기).
    start_c, start_r = int(start_pos[0] // TILE_SIZE), int(start_pos[1] // TILE_SIZE)
    target_c, target_r = int(target_pos[0] // TILE_SIZE), int(target_pos[1] // TILE_SIZE)

    # 마우스로 클릭한 곳이 맵 밖이거나, 하필 꽉 막힌 벽 위를 클릭했다면 경로가 없으므로 빈 리스트를 반환합니다.
    if not (0 <= target_r < grid_rows and 0 <= target_c < grid_cols): return []
    if collision_grid[target_r][target_c]: return []

    open_set = [] # 앞으로 탐색해 볼 칸들의 우선순위 큐(대기열)
    heapq.heappush(open_set, (0, 0, (start_r, start_c))) # 시작점 삽입
    
    came_from = {} # 이 칸에 어디를 거쳐서 왔는지 '발자취'를 기록해두는 딕셔너리 지도
    g_score = {(start_r, start_c): 0} # 시작점부터 해당 칸까지 걸어온 '실제 거리(비용)'

    while open_set:
        # 큐에서 F 점수(총 예상 비용)가 가장 낮은, 즉 가장 유망해 보이는 칸을 하나 뽑아옵니다.
        _, current_g, current = heapq.heappop(open_set)

        # 만약 그 칸이 우리가 찾던 목적지라면 탐색을 대성공으로 종료합니다.
        if current == (target_r, target_c):
            path = []
            curr = current
            # came_from 지도에 남겨진 발자취를 역추적(While)하여 지나온 좌표들을 리스트에 쌓습니다.
            while curr in came_from:
                path.append((curr[1] * TILE_SIZE, curr[0] * TILE_SIZE))
                curr = came_from[curr]
            path.reverse() # 역추적했으므로 순서를 바르게 뒤집어 반환합니다.
            return path

        # 도착지가 아니라면 상하좌우(십자 방향) 4칸을 살펴봅니다.
        cr, cc = current
        for dr, dc in [(-1, 0), (1, 0), (0, -1), (0, 1)]:
            nr, nc = cr + dr, cc + dc
            # 새 칸이 맵 범위 안에 있고 벽이 아니라면 지나갈 수 있습니다.
            if 0 <= nr < grid_rows and 0 <= nc < grid_cols:
                if collision_grid[nr][nc]: continue
                
                # 새 칸까지 이동하는데 드는 비용은 현재 비용 + 1칸 입니다.
                tent_g = current_g + 1
                
                # 이 경로가 예전에 계산해 둔 비용(g_score)보다 더 낫고 빠르다면 정보를 갱신합니다.
                if (nr, nc) not in g_score or tent_g < g_score[(nr, nc)]:
                    g_score[(nr, nc)] = tent_g
                    # H 점수(휴리스틱): 현재 칸에서 목적지까지 남은 칸 수를 빼기(-) 연산으로 예측합니다 (맨해튼 거리).
                    h_score = abs(nr - target_r) + abs(nc - target_c)
                    f_score = tent_g + h_score # F = G + H (최종 평가 점수)
                    
                    # 새로운 정보를 담아 우선순위 큐에 집어넣습니다.
                    heapq.heappush(open_set, (f_score, tent_g, (nr, nc)))
                    came_from[(nr, nc)] = current # 발자취 기록
    
    # 큐를 모두 비웠는데도 길을 찾지 못했다면 빈 리스트 반환 (길이 꽉 막힌 상태)
    return []

running = True
while running:
    for event in pygame.event.get():
        if event.type == pygame.QUIT: running = False
        if event.type == pygame.KEYDOWN:
            # 워프 조준 대기 중이거나 하늘을 비행 중일 때는 키보드 점프 입력을 완전히 차단합니다.
            if not has_warp_ability and not is_flying:
                if event.key == pygame.K_SPACE and is_grounded:
                    velocity_y = jump_power
                    is_grounded = False
            if event.key == pygame.K_r:
                reset_stage()

        # [마우스 입력 이벤트 감지]
        # 왼쪽 마우스 버튼(1)을 클릭했을 때 워프 대기 상태(has_warp_ability)라면 비행을 개시합니다.
        if event.type == pygame.MOUSEBUTTONDOWN and event.button == 1:
            if has_warp_ability and not is_flying:
                cam_x = max(0, player.centerx - SCREEN_WIDTH // 2)
                # 마우스 커서의 화면상 픽셀 좌표에 카메라 이동량을 더해, 맵 상의 실제(World) 목적지 좌표를 역산합니다.
                mouse_w_x = event.pos[0] + cam_x
                mouse_w_y = event.pos[1]
                
                # 출발지와 목적지 좌표를 던져주어 A* 알고리즘의 비행 경로를 알아옵니다.
                path = find_a_star_path((player.x, player.y), (mouse_w_x, mouse_w_y))
                if path:
                    fly_path = path       # 계산된 리스트를 가져와 담음
                    is_flying = True      # 이제부터 경로를 따라가도록 비행 상태 스위치 온
                    has_warp_ability = False # 비행이 시작되었으므로 조준 모드 해제

    # --- 상태별 물리 및 이동 처리 ---
    if is_flying:
        # [상태 1: A* 자동 비행 모드] 
        # 중력을 완전히 무시하고 배열(fly_path)에 든 노드(꺾이는 지점)들을 하나씩 빼가며 순차 추적합니다.
        if fly_path:
            target_node = fly_path[0] # 이번에 도착해야 할 목표 노드
            dx = target_node[0] - player.x
            dy = target_node[1] - player.y
            dist = (dx**2 + dy**2) ** 0.5 # 피타고라스 정리로 거리를 구합니다.
            
            # 목표 지점에 거의 도달했다면(비행 속도 안쪽으로 좁혀짐) 노드를 팝(Pop)하여 다음 목표로 갱신합니다.
            if dist < fly_speed:
                player.x, player.y = target_node
                fly_path.pop(0) 
            else:
                # 방향 벡터(비율)에 속도를 곱하여 일정한 속도로 대각선 직선 이동을 구현합니다.
                player.x += (dx / dist) * fly_speed
                player.y += (dy / dist) * fly_speed
        else:
            # 배열이 비워졌다면 목적지에 무사히 도착한 것이므로 보행 모드로 복귀합니다.
            is_flying = False
            velocity_x = 0.0
            velocity_y = 0.0

    elif has_warp_ability:
        # [상태 2: 워프 조준 대기 모드 (공중 정지)]
        # 워프 아이템을 먹으면 마우스를 클릭할 때까지 속도를 0으로 유지하여 중력을 무시하고 공중에 멈춰섭니다.
        velocity_x = 0.0
        velocity_y = 0.0

    else:
        # [상태 3: 일반 보행 모드 (중력 작용)]
        keys = pygame.key.get_pressed()
        velocity_x = 0
        if keys[pygame.K_LEFT]: velocity_x = -speed
        if keys[pygame.K_RIGHT]: velocity_x = speed

        player.x += velocity_x
        for block in platforms:
            if player.colliderect(block):
                if velocity_x > 0: player.right = block.left
                elif velocity_x < 0: player.left = block.right

        velocity_y += gravity
        player.y += velocity_y
        is_grounded = False

        for block in platforms:
            if player.colliderect(block):
                if velocity_y > 0:
                    player.bottom = block.top
                    velocity_y = 0.0
                    is_grounded = True
                elif velocity_y < 0:
                    player.top = block.bottom
                    velocity_y = 0.0

        # 보행 중 주황색 워프 아이템(W)을 먹으면 스위치를 켜고 조준 모드로 전환
        for warp in warp_items[:]:
            if player.colliderect(warp):
                has_warp_ability = True
                warp_items.remove(warp)
                velocity_x = 0.0
                velocity_y = 0.0

        if player.y > SCREEN_HEIGHT + 100:
            reset_stage()

    # --- [상시 충돌 감지] 보행 중, 공중 정지 중, 무시무시한 비행 중 언제라도 가시에 살이 닿으면 즉시 사망 ---
    player_hitbox = get_padded_rect(player, 4)
    for spike in hazards:
        if player_hitbox.colliderect(get_padded_rect(spike, 6)):
            reset_stage()
            break

    # 화면 렌더링
    camera_x = max(0, player.centerx - SCREEN_WIDTH // 2)
    screen.fill((20, 24, 34))

    for block in platforms:
        r = pygame.Rect(block.x - camera_x, block.y, block.width, block.height)
        if -TILE_SIZE <= r.x <= SCREEN_WIDTH: pygame.draw.rect(screen, (41, 128, 185), r)
    for spike in hazards:
        r = pygame.Rect(spike.x - camera_x, spike.y, spike.width, spike.height)
        if -TILE_SIZE <= r.x <= SCREEN_WIDTH: pygame.draw.rect(screen, (231, 76, 60), r)
    for warp in warp_items:
        r = pygame.Rect(warp.x - camera_x, warp.y, warp.width, warp.height)
        pygame.draw.rect(screen, (230, 126, 34), r)
    if goal_rect:
        r = pygame.Rect(goal_rect.x - camera_x, goal_rect.y, goal_rect.width, goal_rect.height)

    if has_warp_ability: pygame.draw.rect(screen, (230, 126, 34), screen_player, 4)
