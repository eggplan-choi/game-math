활동지 01물리 엔진 기초

[활동지 01] 2D 물리 엔진의 기초: 화면과 중력, 그리고 점프

🎯 학습 목표

  1. 컴퓨터 화면의 좌표계와 게임 루프(Game Loop)의 동작 원리를 이해한다.
  2. 중력 가속도 수식을 코드로 모델링하여 캐릭터가 포물선 운동을 하도록 만든다.
  3. 바닥 착지 조건을 제어하여 무한 점프 버그를 방지한다.

1. 게임 루프와 화면 좌표계

컴퓨터 모니터는 수학 시간에 배우는 데카르트 좌표계와 달리 화면의 왼쪽 위가 원점 $(0, 0)$ 입니다.

  • $x$ 축: 오른쪽으로 갈수록 값이 커집니다 ($+$).
  • $y$ 축: 아래로 내려갈수록 값이 커집니다 ($+$).

게임은 멈춰 있는 그림이 아닙니다. "입력 받기 $\rightarrow$ 위치 계산하기 $\rightarrow$ 화면 새로 그리기" 과정을 1초에 60번씩 반복하는 게임 루프(Game Loop) 위에서 돌아갑니다.

📌 Step 1: 기본 창 띄우고 좌우로 움직이기

🐍 step1_basic_loop.py
import pygame
import sys

pygame.init()
SCREEN_WIDTH = 800
SCREEN_HEIGHT = 450
screen = pygame.display.set_mode((SCREEN_WIDTH, SCREEN_HEIGHT))
pygame.display.set_caption("1차시: 플레이어 이동과 기본 루프")
clock = pygame.time.Clock()

player_x = 100
player_y = 200
player_size = 40
speed = 5

running = True
while running:
 for event in pygame.event.get():
 if event.type == pygame.QUIT:
 running = False

 keys = pygame.key.get_pressed()
 if keys[pygame.K_LEFT]:
 player_x -= speed
 if keys[pygame.K_RIGHT]:
 player_x += speed

 player_x = max(0, min(player_x, SCREEN_WIDTH - player_size))

 screen.fill((30, 30, 30))
 pygame.draw.rect(screen, (0, 200, 255), (player_x, player_y, player_size, player_size))
 pygame.display.flip()
 clock.tick(60)

pygame.quit()
sys.exit()

2. 중력 가속도와 점프 (물리 법칙 도입)

$$v_y \leftarrow v_y + g$$
$$y \leftarrow y + v_y$$
🧪 중력 & 점프 포물선 실시간 시뮬레이터 스페이스바 / 클릭으로 점프
중력 가속도 (gravity)0.8
점프 초기 속도 (jump_power)-14.0
수평 이동 속도 (speed)5.0

📌 Step 2: 중력과 점프 구현하기

🐍 step2_gravity_jump.py
import pygame
import sys

pygame.init()
SCREEN_WIDTH = 800
SCREEN_HEIGHT = 450
screen = pygame.display.set_mode((SCREEN_WIDTH, SCREEN_HEIGHT))
pygame.display.set_caption("1차시: 중력과 점프 물리 엔진")
clock = pygame.time.Clock()

player_x = 100
player_y = 100
player_size = 40
speed = 5

velocity_y = 0.0
gravity = 0.8
jump_power = -14.0
is_grounded = False
FLOOR_Y = 380

running = True
while running:
 for event in pygame.event.get():
 if event.type == pygame.QUIT:
 running = False
 if event.type == pygame.KEYDOWN:
 if event.key == pygame.K_SPACE and is_grounded:
 velocity_y = jump_power
 is_grounded = False

 keys = pygame.key.get_pressed()
 if keys[pygame.K_LEFT]: player_x -= speed
 if keys[pygame.K_RIGHT]: player_x += speed
 player_x = max(0, min(player_x, SCREEN_WIDTH - player_size))

 velocity_y += gravity
 player_y += velocity_y

 if player_y + player_size >= FLOOR_Y:
 player_y = FLOOR_Y - player_size
 velocity_y = 0.0
 is_grounded = True

 screen.fill((25, 25, 35))
 pygame.draw.rect(screen, (46, 204, 113), (0, FLOOR_Y, SCREEN_WIDTH, SCREEN_HEIGHT - FLOOR_Y))
 pygame.draw.rect(screen, (230, 126, 34), (player_x, int(player_y), player_size, player_size))
 pygame.display.flip()
 clock.tick(60)

pygame.quit()
sys.exit()

[활동지 02] 사각 충돌(AABB)과 너그러운 판정(Hitbox Inset)

🎯 학습 목표

  1. 두 사각형이 겹치는 수학적 조건(AABB 충돌)을 유도하고 코드로 구현한다.
  2. 축 분리 방식을 통해 발판 착지와 벽 막힘을 정확히 처리한다.
  3. 렌더링 박스와 피격 판정 박스를 분리(패딩 적용)하여 불합리하지 않은 난이도를 설계한다.

1. 사각형은 언제 부딪힐까? (AABB 충돌 판정)

$$x_1 < x_2 + w_2 \quad \text{and} \quad x_1 + w_1 > x_2$$
$$y_1 < y_2 + h_2 \quad \text{and} \quad y_1 + h_1 > y_2$$
🐍 step2_hazard_padding.py
import pygame
import sys

pygame.init()
SCREEN_WIDTH = 800
SCREEN_HEIGHT = 450
screen = pygame.display.set_mode((SCREEN_WIDTH, SCREEN_HEIGHT))
pygame.display.set_caption("2차시: 히트박스 패딩과 사망/리스폰")
clock = pygame.time.Clock()

SPAWN_POINT = (60, 300)
player = pygame.Rect(SPAWN_POINT[0], SPAWN_POINT[1], 40, 40)
speed = 5
velocity_y = 0.0
gravity = 0.8
jump_power = -14.0
is_grounded = False

PLAYER_PAD = 4
HAZARD_PAD = 6
debug_mode = False

platforms = [
 pygame.Rect(0, 380, 800, 70),
 pygame.Rect(220, 290, 140, 20),
 pygame.Rect(440, 230, 140, 20),
]
hazards = [
 pygame.Rect(270, 260, 30, 30),
 pygame.Rect(500, 350, 40, 30),
]

def get_padded_rect(rect, pad):
 return pygame.Rect(rect.x + pad, rect.y + pad, rect.width - 2 * pad, rect.height - 2 * pad)

running = True
while running:
 for event in pygame.event.get():
 if event.type == pygame.QUIT:
 running = False
 if event.type == pygame.KEYDOWN:
 if event.key == pygame.K_SPACE and is_grounded:
 velocity_y = jump_power
 is_grounded = False
 if event.key == pygame.K_F1:
 debug_mode = not debug_mode

 keys = pygame.key.get_pressed()
 velocity_x = 0
 if keys[pygame.K_LEFT]: velocity_x = -speed
 if keys[pygame.K_RIGHT]: velocity_x = speed

 player.x += velocity_x
 for block in platforms:
 if player.colliderect(block):
 if velocity_x > 0: player.right = block.left
 elif velocity_x < 0: player.left = block.right

 velocity_y += gravity
 player.y += int(velocity_y)
 is_grounded = False

 for block in platforms:
 if player.colliderect(block):
 if velocity_y > 0:
 player.bottom = block.top
 velocity_y = 0.0
 is_grounded = True
 elif velocity_y < 0:

def generate_beep_sound(freq_start, freq_end, duration_ms, wave_type="square"):
 sample_rate = 44100
 n_samples = int(sample_rate * (duration_ms / 1000.0))
 buf = array.array('h')
 for i in range(n_samples):
 t = i / n_samples
 freq = freq_start + (freq_end - freq_start) * t
 phase = 2.0 * math.pi * freq * (i / sample_rate)
 val = 16000 if math.sin(phase) >= 0 else -16000
 decay = (1.0 - t)
 buf.append(int(val * decay))

 if collision_grid[target_r][target_c]: return []

 heapq.heappush(open_set, (0, 0, (start_r, start_c)))
 came_from = {}
 g_score = {(start_r, start_c): 0}

 while open_set:
 _, current_g, current = heapq.heappop(open_set)
 if current == (target_r, target_c):
 path = []
 curr = current
 while curr in came_from:
 path.append((curr[1] * TILE_SIZE, curr[0] * TILE_SIZE))
 curr = came_from[curr]
 path.reverse()
 return path

 for dr, dc in [(-1, 0), (1, 0), (0, -1), (0, 1)]:

 g_score[(nr, nc)] = tent_g
 f_score = tent_g + abs(nr - target_r) + abs(nc - target_c)
 heapq.heappush(open_set, (f_score, tent_g, (nr, nc)))
 came_from[(nr, nc)] = current
 return []
