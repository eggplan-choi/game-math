[활동지 01] 아두이노 환경 구축 및 오픈 프레임 프로토타입 조립

1부에서 소프트웨어로 구현했던 게임 캐릭터를 현실 세계의 아날로그 조이스틱과 독립 푸시 버튼으로 제어하기 위해, 하드웨어를 직접 결선하고 파이썬 시리얼 드라이버를 준비합니다.

2. 역논리(Negative Logic)와 소프트웨어 보정

위 [코드 2]를 실행해 보면, 버튼을 가만히 두었을 때는 1이 나오고, 버튼을 꾹 누르는 순간 전류가 접지로 빠져나가며 0이 출력됩니다. 우리가 일상적으로 생각하는 '누름 = 1'이라는 직관과 반대되는 현상입니다.

💡 논리 부정 연산자 `!` (NOT)

이러한 역논리 현상을 가장 쉽게 해결하는 방법은 프로그래밍의 논리 부정 연산자인 `!` (NOT)를 사용하는 것입니다. 값 앞에 !를 붙이면 0은 1로, 1은 0으로 완벽하게 뒤집어줍니다.

$$\text{btn\_jump} = !\text{digitalRead(PIN\_BTN)} \quad \iff \quad \text{btn\_jump} = 1 - \text{digitalRead(PIN\_BTN)}$$

수학적으로 ! 기호를 붙이는 것은 1 - 현재값을 계산하는 것과 완벽히 동일한 결과($1-1=0$, $1-0=1$)를 만들어냅니다.

  2. 시리얼 모니터를 켜고 버튼을 누를 때 Corrected 값이 우리가 원하는 대로 1 (점프)로 출력되는지 확인합니다.

const int PIN_BTN = 2; // 디지털 2번 핀

void setup() {
  Serial.begin(115200); 
  pinMode(PIN_BTN, INPUT_PULLUP); // 내부 풀업 저항 켜기
}

void loop() {
  int raw_signal = digitalRead(PIN_BTN);
  
  // ! 연산자를 통한 논리 반전 (0을 1로, 1을 0으로)
  int corrected_signal = !raw_signal; 

  Serial.print("Raw: ");
  Serial.print(raw_signal);
  Serial.print("  |  Corrected: ");
  Serial.println(corrected_signal);

  delay(50);
}

1. ADC(아날로그-디지털 변환)와 데드존($D$)

조이스틱은 기울인 각도에 따라 가변저항 전압이 변하며, 아두이노는 이 전압 신호를 10비트 해상도를 통해 아래 부등식 범위의 정수로 환산합니다.

$$0 \le \text{raw} \le 1023, \quad \text{중립 중심값} \approx 512$$

2. 정규화 수식의 단계별 유도

조이스틱의 값을 우리가 게임 속도로 활용하기 편한 $-1.0 \sim +1.0$ 비율로 바꾸는 과정을 정규화(Normalization)라고 합니다. 기본적인 수학 공식은 다음과 같습니다.

$$ \text{비율} = \frac{\text{현재값} - \text{시작점(최솟값)}}{\text{끝점(최댓값)} - \text{시작점(최솟값)}} $$

수식 유도 1: 양의 방향으로 밀 때 ($\text{raw} > 512 + D$)

  • 분자에서 왜 $512 + D$를 뺄까? 데드존 경계를 넘어선 순간부터 이동이 시작되어야 하므로 조작 시작점을 $0$으로 맞추기 위함입니다.
  • 분모는 왜 $511 - D$ 일까? 스틱을 끝까지 밀었을 때 최댓값은 $1023$입니다. 분자($1023 - (512+D)$)가 분모 전체 길이와 동일해야 정규화 비율이 완벽하게 $+1.0$에 도달합니다.
$$\text{tilt\_x} = \frac{\text{raw} - (512 + D)}{511 - D} \quad (0.0 \le \text{tilt\_x} \le 1.0)$$

수식 유도 2: 음의 방향으로 밀 때 ($\text{raw} < 512 - D$)

$$\text{tilt\_x} = \frac{\text{raw} - (512 - D)}{512 - D} \quad (-1.0 \le \text{tilt\_x} \le 0.0)$$

  2. 시리얼 모니터를 열고 스틱을 돌려보세요. 0.85,-0.60,1 처럼 3개의 값이 쉼표로 묶여 실시간으로 깔끔하게 올라오는 것을 확인합니다.
⚡ [아두이노 C++] step2_joystick_packet.ino 
const int PIN_JOY_X = A0; 
const int PIN_JOY_Y = A1; 
const int PIN_BTN = 2;    

const int CENTER = 512;
const int D = 70; // 데드존 D

void setup() {
  Serial.begin(115200); 
  pinMode(PIN_BTN, INPUT_PULLUP);
}

// 수학 공식을 함수로 묶어 깔끔하게 재사용!
float get_tilt(int raw) {
  if (raw > CENTER + D) return (float)(raw - (CENTER + D)) / (1023 - (CENTER + D));
  if (raw < CENTER - D) return (float)(raw - (CENTER - D)) / (CENTER - D);
  return 0.0;
}

void loop() {
  float tilt_x = get_tilt(analogRead(PIN_JOY_X));
  float tilt_y = get_tilt(analogRead(PIN_JOY_Y)); // Y축도 동일한 공식 적용

  // 범위를 완벽하게 -1.0 ~ 1.0으로 가두기 (클램핑)
  if (tilt_x > 1.0) tilt_x = 1.0; if (tilt_x < -1.0) tilt_x = -1.0;
  if (tilt_y > 1.0) tilt_y = 1.0; if (tilt_y < -1.0) tilt_y = -1.0;

  int btn_jump = !digitalRead(PIN_BTN); // 2차시에서 배운 논리 보정 적용

  // 직렬화 전송: "tilt_x,tilt_y,btn_jump\n" (예: 0.65,-0.20,1)
  Serial.print(tilt_x, 2); Serial.print(",");
  Serial.print(tilt_y, 2); Serial.print(",");
  Serial.println(btn_jump);

  delay(16); // 게임 프레임(60FPS)과 맞춘 16ms 대기
}

1. 입력 지연 방지를 위한 버퍼 플러시 (Queue Flush)

아두이노가 초당 60회 전송할 때, 파이썬이 단순히 readline()을 1회만 호출하면 어떻게 될까요? 통신 큐(Queue)에 과거 데이터가 처리되지 못하고 쌓이면서 조작 반응이 0.5초씩 밀리게 됩니다.

$$\text{Buffer Flush}: \quad \text{while}(ser.in\_waiting > 0) \implies \text{latest\_packet} \leftarrow \text{readline}()$$

def generate_beep(freq_start, freq_end, duration_ms, wave_type="square"):
    sample_rate = 44100
    n_samples = int(sample_rate * (duration_ms / 1000.0))
    buf = array.array('h')
    for i in range(n_samples):
        t = i / n_samples
        freq = freq_start + (freq_end - freq_start) * t
        phase = 2.0 * math.pi * freq * (i / sample_rate)
        if wave_type == "square": val = 14000 if math.sin(phase) >= 0 else -14000
        elif wave_type == "noise": val = random.randint(-14000, 14000)
        else: val = int(14000 * math.sin(phase))
        buf.append(int(val * (1.0 - t)))
    return pygame.mixer.Sound(buf)

class Particle:
    def __init__(self, x, y, color):
        self.x, self.y, self.color = float(x), float(y), color
        self.size = random.uniform(6.0, 11.0)
        
        # [수학적 원리] 극좌표계(r, θ)를 활용한 파티클 분사
        # 무작위 각도(angle)와 속력(speed)을 생성한 뒤, 
        # 삼각비(cos, sin)를 곱해 직교 좌표계의 x, y 속도 벡터(vx, vy)로 변환합니다.
        angle = random.uniform(0, 2 * math.pi)
        speed_val = random.uniform(2.5, 7.5)
        self.vx = math.cos(angle) * speed_val
        self.vy = math.sin(angle) * speed_val - 2.0
        
        self.gravity, self.life, self.decay = 0.35, 1.0, random.uniform(0.025, 0.045)

    def update(self):
        self.vy += self.gravity  # 중력 가속도 누적 (y방향 속도 증가)
        self.x += self.vx        # 수평 위치 갱신
        self.y += self.vy        # 수직 위치 갱신
        self.life -= self.decay
        self.size = max(0.0, self.size - 0.15)

    def draw(self, surface, cam_x, ox, oy):
        if self.life > 0 and self.size > 0:
            rx, ry = int(self.x - cam_x + ox), int(self.y + oy)
            pygame.draw.rect(surface, self.color, (rx, ry, int(self.size), int(self.size)))

def find_a_star_path(start_pos, target_pos):
    """
    [수학적 원리] A* (A-Star) 최단 경로 탐색 알고리즘
    f(n) = g(n) + h(n) 공식을 사용하여 목적지까지의 가장 짧은 경로를 찾습니다.
    g(n): 시작점에서 현재까지 이동한 칸 수
    h(n): 현재 위치에서 목적지까지의 맨해튼 거리 (가로 칸 수 차이 + 세로 칸 수 차이)
    """
    sc, sr = int(start_pos[0] // TILE_SIZE), int(start_pos[1] // TILE_SIZE)
    tc, tr = int(target_pos[0] // TILE_SIZE), int(target_pos[1] // TILE_SIZE)
    
    # 맵 바깥이거나 도착지가 막힌 벽이면 탐색 포기
    if not (0 <= tr < MAP_ROWS and 0 <= tc < MAP_COLS) or collision_grid[tr][tc]: return []

    open_set, came_from, g_score = [], {}, {(sr, sc): 0}
    heapq.heappush(open_set, (0, 0, (sr, sc)))

    while open_set:
        _, cur_g, current = heapq.heappop(open_set)
        if current == (tr, tc):
            path, curr = [], current
            while curr in came_from:
                # 타일 모서리가 아닌 중심점(+20픽셀)을 지나는 궤적으로 보정
                path.append((curr[1] * TILE_SIZE + 20, curr[0] * TILE_SIZE + 20))
                curr = came_from[curr]
            path.reverse()
            return path

        cr, cc = current
        # 상하좌우 4방향 탐색
        for dr, dc in [(-1, 0), (1, 0), (0, -1), (0, 1)]:
            nr, nc = cr + dr, cc + dc
            if 0 <= nr < MAP_ROWS and 0 <= nc < MAP_COLS and not collision_grid[nr][nc]:
                tentative_g = cur_g + 1
                if (nr, nc) not in g_score or tentative_g < g_score[(nr, nc)]:
                    g_score[(nr, nc)] = tentative_g
                    f = tentative_g + abs(nr - tr) + abs(nc - tc) # 맨해튼 거리 휴리스틱 추가
                    heapq.heappush(open_set, (f, tentative_g, (nr, nc)))
                    came_from[(nr, nc)] = current
    return []

        pygame.draw.rect(world_surface, (230, 126, 34), (rx, ry, TILE_SIZE, TILE_SIZE), 3)

    # 역주행 상태일 때 화면 전체를 상하 반전시키는 수학적 렌더 변환
    if is_reversed:
        flipped_world = pygame.transform.flip(world_surface, False, True)
        screen.blit(flipped_world, (0, 0))
    else:
        screen.blit(world_surface, (0, 0))

    # ==========================================================================
    # 9. 조준점(크로스헤어) 및 상태 HUD 렌더링 (화면 픽셀계에 고정)
    # ==========================================================================
    if has_warp_ability:
        cx, cy = int(cursor_x), int(cursor_y)
        pygame.draw.line(screen, (255, 50, 50), (cx - 15, cy), (cx + 15, cy), 3)
        pygame.draw.line(screen, (255, 50, 50), (cx, cy - 15), (cx, cy + 15), 3)
        pygame.draw.circle(screen, (255, 50, 50), (cx, cy), 10, 2)
        
        prompt = font_bold.render("조이스틱으로 조준 후 [점프 버튼]을 눌러 워프하세요!", True, (255, 240, 0))
        p_bg = pygame.Rect(SCREEN_WIDTH // 2 - prompt.get_width() // 2 - 14, 60, prompt.get_width() + 28, prompt.get_height() + 12)
        pygame.draw.rect(screen, (10, 14, 24), p_bg)
        pygame.draw.rect(screen, (230, 126, 34), p_bg, 2)
        screen.blit(prompt, (SCREEN_WIDTH // 2 - prompt.get_width() // 2, 66))

    if troll_timer > 0:
        t_msg = font_large.render("FAKE GOAL! TRAP ACTIVATED!", True, (255, 60, 60))
        screen.blit(t_msg, (SCREEN_WIDTH // 2 - t_msg.get_width() // 2, 110))

    if reverse_announcement_timer > 0:
        r_msg = font_large.render("GRAVITY INVERTED! ESCAPE BACK TO START!", True, (190, 100, 255))
        screen.blit(r_msg, (SCREEN_WIDTH // 2 - r_msg.get_width() // 2, 110))

    if is_game_cleared:
        c_msg = font_large.render("ALL MISSIONS CLEARED! YOU ESCAPED!", True, (46, 204, 113))
        screen.blit(c_msg, (SCREEN_WIDTH // 2 - c_msg.get_width() // 2, SCREEN_HEIGHT // 2 - 20))

    mode_str = "REVERSED (RETURN!)" if is_reversed else "FORWARD (TO GOAL)"
    hud_progress = f"Pos: {int(player_x // TILE_SIZE)}m | Tilt: {hw_tilt_x:+.2f} | Deaths: {death_count} | {mode_str}"
    t_hud1 = font_ui.render(hud_progress, True, (220, 220, 220))
    screen.blit(t_hud1, (20, 15))

    pygame.display.flip()
    clock.tick(60)

if ser and ser.is_open:
    ser.close()
pygame.quit()
sys.exit()
